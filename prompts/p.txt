common-script.js:
// Script for handling navigation menu toggle and active item highlighting

const menuControl = document.getElementById('menu-control');
const navigationMenu = document.getElementById('navigation-menu');
const mainContent = document.getElementById('main-content');
let navItems = [];

// Toggle menu function
const toggleMenu = () => {
    navigationMenu.classList.toggle('open');
    mainContent.classList.toggle('shifted');
    
    menuControl.style.opacity = '0';
    menuControl.style.transform = 'rotate(90deg)';

    setTimeout(() => {
        if (menuControl.classList.contains('fa-bars')) {
            menuControl.classList.remove('fa-bars');
            menuControl.classList.add('fa-arrow-left');
        } else {
            menuControl.classList.remove('fa-arrow-left');
            menuControl.classList.add('fa-bars');
        }

        menuControl.style.opacity = '1';
        menuControl.style.transform = 'rotate(0deg)';
    }, 200);
};

// Add click event to menu control
menuControl.addEventListener('click', toggleMenu);

// Close menu when clicking outside on mobile
document.addEventListener('click', (event) => {
    if (window.innerWidth <= 768 && 
        navigationMenu.classList.contains('open') &&
        !event.target.closest('.navigation-menu') &&
        !event.target.closest('#menu-control')) {
        toggleMenu();
    }
});

// Handle window resize
window.addEventListener('resize', () => {
    if (window.innerWidth > 768) {
        navigationMenu.classList.add('open');
        mainContent.classList.add('shifted');
        menuControl.classList.remove('fa-bars');
        menuControl.classList.add('fa-arrow-left');
    } else {
        navigationMenu.classList.remove('open');
        mainContent.classList.remove('shifted');
        menuControl.classList.remove('fa-arrow-left');
        menuControl.classList.add('fa-bars');
    }
});

// Initialize based on screen size
if (window.innerWidth > 768) {
    navigationMenu.classList.add('open');
    mainContent.classList.add('shifted');
    menuControl.classList.remove('fa-bars');
    menuControl.classList.add('fa-arrow-left');
}

// Script for dynamically loading navigation menu items from the HTTP API
const navList = document.getElementById('nav-list');
const authToken = localStorage.getItem('authToken');

// Fetch pages via HTTP API with Authorization header
fetch('/api/pages', {
    headers: {
        'Authorization': `Bearer ${authToken}`
    }
})
.then(response => {
    if (!response.ok) {
        throw new Error(`HTTP error! Status: ${response.status}`);
    }
    return response.json();
})
.then(data => {
    navList.innerHTML = ""; // clear menu

    data.pages.forEach((page, index) => {
        const li = document.createElement('li');
        li.classList.add('nav-item');

        const currentPath = window.location.pathname.replace('/', '');
        const pagePath = page.file.replace('.html', '');

        if (currentPath === pagePath || (currentPath === '' && pagePath === 'dashboard')) {
            li.classList.add('active');
        }

        // Get badge count from API - ONLY use what's in the API
        const badgeCount = page.badge || 0;
        
        // Create badge HTML
        let badgeHtml = '';
        if (badgeCount > 0) {
            const badgeText = badgeCount > 99 ? '99+' : badgeCount;
            const pulseClass = badgeCount > 5 ? ' pulse' : '';
            badgeHtml = `<span class="nav-badge${pulseClass}">${badgeText}</span>`;
        } else {
            badgeHtml = '<span class="nav-badge hidden">0</span>';
        }

        li.innerHTML = `
            <i class="fa ${page.icon}"></i>
            <p>${page.name}</p>
            ${badgeHtml}
        `;

        // Click handler
        li.addEventListener('click', () => {
            document.querySelectorAll('.nav-item')
                .forEach(item => item.classList.remove('active'));

            li.classList.add('active');

            // Navigate using clean URL
            const pageName = page.file.replace(/\.html$/, '');
            window.location.href = `/${pageName}`;
        });

        navList.appendChild(li);
    });
    
    navItems = document.querySelectorAll('.nav-item');
})
.catch(err => {
    console.error('Error fetching pages:', err);
    navList.innerHTML = "<li class='nav-item'>Failed to load menu</li>";
});
